```kotlin
package com.streamy.app.audio

/** Pure DSP helpers for the live VU meter: RMS level (dB) + decimation of a PCM frame. */
object LevelTap {
    private const val FLOOR_DB = -90.0f

    /** RMS level in dBFS (0 = full scale, silence clamped to -90). */
    fun rmsDb(pcm: ShortArray): Float {
        var sum = 0.0
        for (s in pcm) {
            val n = s.toDouble() / 32768.0
            sum += n * n
        }
        val rms = kotlin.math.sqrt(sum / pcm.size)
        val db = (20.0 * kotlin.math.log10(kotlin.math.max(rms, 1e-7))).toFloat()
        return if (db < FLOOR_DB) FLOOR_DB else db
    }

    /** Evenly decimate [pcm] to [points] samples, normalized to -1..1. */
    fun decimate(pcm: ShortArray, points: Int): FloatArray {
        val out = FloatArray(points)
        if (pcm.isEmpty()) return out
        for (i in 0 until points) {
            val src = (i * (pcm.size - 1)) / (points - 1).coerceAtLeast(1)
            out[i] = pcm[src].toFloat() / 32768.0f
        }
        return out
    }
}
```

**Step 1: Implement** (no unit test — it's a thin StateFlow holder; exercised via the running app).

```kotlin
package com.streamy.app.audio

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

/** One meter frame: a decimated, normalized waveform snapshot + its RMS level. */
class LevelSnapshot(val samples: FloatArray, val rmsDb: Float)

/**
 * Holds the latest [LevelSnapshot] for the UI. Producers (the audio thread) call [push];
 * the UI collects [flow]. A ~30 fps time gate caps decimation/GC cost — [StateFlow] itself
 * is conflated, so a slow UI just sees the latest frame and never blocks the audio thread.
 */
class LevelFeed {
    private val _flow = MutableStateFlow<LevelSnapshot?>(null)
    val flow: StateFlow<LevelSnapshot?> = _flow.asStateFlow()

    @Volatile private var lastPushNs = 0L

    fun push(snapshot: LevelSnapshot, nowNs: Long) {
        if (nowNs - lastPushNs < THROTTLE_NS) return
        lastPushNs = nowNs
        _flow.value = snapshot
    }

    fun reset() { _flow.value = null }

    companion object { private const val THROTTLE_NS = 33_000_000L } // ~30 fps
}
```
